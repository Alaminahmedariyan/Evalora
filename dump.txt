import type { Request, Response } from "express";
import { StatusCodes } from "http-status-codes";

import { catchAsync } from "../../utils/catchAsync";
import { sendResponse } from "../../utils/sendResponse";

import { automationAi } from "./automation.ai";
import { automationMaintenance } from "./automation.maintenance";
import { automationService } from "./automation.service";

const dispatch = catchAsync(async (_req: Request, res: Response) => {
	const result = await automationService.dispatchPendingEvents();

	sendResponse(res, {
		success: true,
		statusCode: StatusCodes.OK,
		message: "Automation events dispatched.",
		data: result,
	});

	const result = await automationMaintenance.runMaintenance();

	sendResponse(res, {
		success: true,
		statusCode: StatusCodes.OK,
		message: "Maintenance run completed.",
		data: result,
	});

		success: true,

===== src/app/modules/automation/automation.maintenance.ts =====
import { prisma } from "../../../lib/prisma";
import { escapeHtml } from "../../utils/escapeHtml";
import { sendEmail } from "../../utils/sendEmail";
import { attemptService } from "../attempt/attempt.service";

const HOUR_MS = 60 * 60 * 1000;
const REMINDER_WINDOW_MS = 24 * HOUR_MS;
const BATCH_SIZE = 50;

const reminderEmailTemplate = (assessmentTitle: string, expiresAt: Date) => `
	<div style="font-family: sans-serif; max-width: 480px; margin: 0 auto; color: #111;">
		<h2>Your invitation expires soon</h2>
		<p>Your invitation to <strong>${escapeHtml(assessmentTitle)}</strong> expires within 24 hours.</p>
		<p>Log in to your account and accept the invitation to start the assessment.</p>
		<p style="font-size: 13px; color: #666;">Expires at: ${expiresAt.toUTCString()}</p>
	</div>
`;

/**
 * Finalizes attempts whose time ran out but nobody submitted or opened.
 * getAttemptById already auto-submits an expired IN_PROGRESS attempt, so
 * this reuses that single code path instead of grading separately.
 */
const expireStaleAttempts = async () => {
	const stale = await prisma.assessmentAttempt.findMany({
		where: { status: "IN_PROGRESS", expiresAt: { lt: new Date() } },
		select: { id: true, candidateId: true },
		orderBy: { expiresAt: "asc" },
		take: BATCH_SIZE,
	});

	let finalized = 0;
	let failed = 0;

	for (const attempt of stale) {
		try {
			await attemptService.getAttemptById(attempt.id, {
				id: attempt.candidateId,
				role: "CANDIDATE",
			});
			finalized += 1;
		} catch (error) {
			failed += 1;
			console.error("[Maintenance] Failed to finalize attempt", attempt.id, error);
		}
	}

	return { found: stale.length, finalized, failed };
};

/** PENDING invitations past their expiry become EXPIRED. */
const expireInvitations = async () => {
	const result = await prisma.assessmentInvitation.updateMany({
		where: { status: "PENDING", expiresAt: { lt: new Date() } },
		data: { status: "EXPIRED" },
	});

	return { expired: result.count };
};

/**
 * Sends one reminder per invitation, 24 hours before it expires.
 * reminderSentAt is claimed BEFORE sending, so two overlapping runs can
 * never email the same candidate twice. If sending fails, the claim is
 * released so the next run retries.
 */
const sendInvitationReminders = async () => {
	const now = new Date();

	const due = await prisma.assessmentInvitation.findMany({
		where: {
			status: "PENDING",
			reminderSentAt: null,
			expiresAt: {
				gt: now,
				lte: new Date(now.getTime() + REMINDER_WINDOW_MS),
			},
			assessment: { deletedAt: null, status: { in: ["PUBLISHED", "ACTIVE"] } },
		},
		select: {
			id: true,
			email: true,
			candidateId: true,
			assessmentId: true,
			expiresAt: true,
			assessment: { select: { title: true } },
		},
		orderBy: { expiresAt: "asc" },
		take: BATCH_SIZE,
	});

	let sent = 0;
	let failed = 0;

	for (const invitation of due) {
		if (!invitation.expiresAt) continue;

		const claimed = await prisma.assessmentInvitation.updateMany({
			where: { id: invitation.id, reminderSentAt: null },
			data: { reminderSentAt: now },
		});

		if (claimed.count === 0) continue; // another run already took it

		try {
			await sendEmail({
				to: invitation.email,
				subject: `Reminder: ${invitation.assessment.title} expires soon`,
				html: reminderEmailTemplate(
					invitation.assessment.title,
					invitation.expiresAt,
				),
			});

			if (invitation.candidateId) {
				try {
					await prisma.notification.create({
						data: {
							userId: invitation.candidateId,
							title: "Invitation expires soon",
							message: `Your invitation to "${invitation.assessment.title}" expires within 24 hours.`,
							type: "ASSESSMENT_REMINDER" as const,
							metadata: {
								assessmentId: invitation.assessmentId,
								invitationId: invitation.id,
							},
						},
					});
				} catch (error) {
					console.error("[Maintenance] Failed to create reminder notification", error);
				}
			}

			sent += 1;
		} catch (error) {
			failed += 1;
			console.error("[Maintenance] Reminder failed", invitation.id, error);

			await prisma.assessmentInvitation.updateMany({
				where: { id: invitation.id },
				data: { reminderSentAt: null },
			});
		}
	}

	return { found: due.length, sent, failed };
};

const runMaintenance = async () => {
	// Order matters: expire first, so an expired invitation never gets a reminder.
	const attempts = await expireStaleAttempts();
	const invitations = await expireInvitations();
	const reminders = await sendInvitationReminders();

	return { attempts, invitations, reminders };
};

export const automationMaintenance = {
	expireStaleAttempts,
	expireInvitations,
	sendInvitationReminders,
	runMaintenance,
};

===== src/app/modules/automation/automation.ai.ts =====
import { StatusCodes } from "http-status-codes";

import { prisma } from "../../../lib/prisma";
import AppError from "../../errors/appError";

import { AUTOMATION_EVENT_TYPES } from "./automation.const";
import { automationService } from "./automation.service";

type AiResultInput = {
	aiScore: number;
	aiFeedback: string;
	aiModel: string;
};

const loadWrittenSubmission = async (submissionId: string) => {
	const submission = await prisma.submission.findUnique({
		where: { id: submissionId },
		select: {
			id: true,
			attemptId: true,
			problemId: true,
			answerText: true,
			evaluation: {
				select: { id: true, status: true, maxScore: true, reviewStatus: true },
			},
			problem: {
				select: { title: true, description: true, type: true, rubric: true },
			},
			attempt: {
				select: {
					candidateId: true,
					assessmentId: true,
					assessment: { select: { companyId: true } },
				},
			},
		},
	});

	if (!submission) {
		throw new AppError(StatusCodes.NOT_FOUND, "Submission not found.");
	}

	if (submission.problem.type !== "WRITTEN") {
		throw new AppError(
			StatusCodes.BAD_REQUEST,
			"AI evaluation is only available for WRITTEN problems.",
		);
	}

	return submission;
};

/**
 * What n8n needs to grade one written answer. No name, email or phone is
 * ever returned. If the candidate has not granted AI_PROCESSING consent
 * (or the submission is already graded), eligible is false and n8n must
 * stop: the submission simply stays PENDING for the recruiter.
 */
const getAiInput = async (submissionId: string) => {
	const submission = await loadWrittenSubmission(submissionId);

	if (!submission.evaluation || submission.evaluation.status !== "PENDING") {
		return { eligible: false as const, reason: "ALREADY_GRADED_OR_NOT_PENDING" };
	}

	if (
		submission.evaluation.reviewStatus === "APPROVED" ||
		submission.evaluation.reviewStatus === "OVERRIDDEN"
	) {
		return { eligible: false as const, reason: "ALREADY_REVIEWED" };
	}

	if (!submission.answerText || submission.answerText.trim() === "") {
		return { eligible: false as const, reason: "EMPTY_ANSWER" };
	}

	const consent = await prisma.userConsent.findUnique({
		where: {
			userId_consentType: {
				userId: submission.attempt.candidateId,
				consentType: "AI_PROCESSING",
			},
		},
		select: { granted: true },
	});

	if (!consent?.granted) {
		return { eligible: false as const, reason: "NO_AI_CONSENT" };
	}

	return {
		eligible: true as const,
		submissionId: submission.id,
		question: {
			title: submission.problem.title,
			description: submission.problem.description,
		},
		rubric: submission.problem.rubric ?? null,
		answer: submission.answerText,
		maxScore: submission.evaluation.maxScore,
	};
};

/**
 * Stores the AI suggestion. It never touches score or status, so the
 * submission stays PENDING and only a recruiter can finalize it.
 */
const saveAiResult = async (submissionId: string, input: AiResultInput) => {
	const submission = await loadWrittenSubmission(submissionId);

	if (!submission.evaluation) {
		throw new AppError(StatusCodes.CONFLICT, "Submission has no evaluation yet.");
	}

	if (
		submission.evaluation.status !== "PENDING" ||
		submission.evaluation.reviewStatus === "APPROVED" ||
		submission.evaluation.reviewStatus === "OVERRIDDEN"
	) {
		throw new AppError(
			StatusCodes.CONFLICT,
			"This submission was already graded by a recruiter.",
		);
	}

	if (input.aiScore > submission.evaluation.maxScore) {
		throw new AppError(
			StatusCodes.BAD_REQUEST,
			`aiScore cannot exceed the maximum marks (${submission.evaluation.maxScore}).`,
		);
	}

	await prisma.submissionEvaluation.update({
		where: { id: submission.evaluation.id },
		data: {
			aiScore: input.aiScore,
			aiFeedback: input.aiFeedback,
			aiModel: input.aiModel,
			aiEvaluatedAt: new Date(),
			reviewStatus: "PENDING_REVIEW",
		},
	});

	const companyId = submission.attempt.assessment.companyId;

	// Best-effort: the AI result is saved even if these two fail.
	try {
		const company = await prisma.company.findUnique({
			where: { id: companyId },
			select: { ownerId: true },
		});

		if (company) {
			await prisma.notification.create({
				data: {
					userId: company.ownerId,
					title: "AI suggestion ready for review",
					message: "A written answer has an AI-suggested score waiting for your review.",
					type: "EVALUATION_REVIEW_NEEDED",
					metadata: {
						submissionId,
						attemptId: submission.attemptId,
						assessmentId: submission.attempt.assessmentId,
					},
				},
			});
		}
	} catch (error) {
		console.error("[AI] Failed to create review notification", error);
	}

	try {
		await automationService.emitEvent({
			eventType: AUTOMATION_EVENT_TYPES.EVALUATION_REVIEW_NEEDED,
			aggregateType: "Submission",
			aggregateId: submissionId,
			dedupeKey: `${AUTOMATION_EVENT_TYPES.EVALUATION_REVIEW_NEEDED}:${submissionId}`,
			payload: {
				submissionId,
				attemptId: submission.attemptId,
				assessmentId: submission.attempt.assessmentId,
			},
		});
	} catch (error) {
		console.error("[AI] Failed to emit review event", error);
	}

	return { saved: true, reviewStatus: "PENDING_REVIEW" as const };
};

export const automationAi = {
	getAiInput,
	saveAiResult,
};

/**

 * human (CODING/WRITTEN with a real answer), across all candidates.

const getPendingEvaluations = async (

	companyId: string,

/**
 * Manually grades a CODING or WRITTEN submission. MCQ submissions are
 * rejected outright - they're graded automatically the moment the attempt
 * is finalized (see grading.util.ts) and re-grading them manually would
 * let a recruiter silently override an objectively-checkable answer.
 *
 * If an AI suggestion was waiting for review, grading it here marks it
 * APPROVED (same score as the AI) or OVERRIDDEN (different score).
 *
 * Triggers recomputeResult() afterward, which is what actually moves the
 * attempt's Result out of PENDING once every submission is COMPLETED.
 */
const evaluateSubmission = async (
	id: string,
	evaluatorId: string,
	companyId: string,
	payload: ManualEvaluationInput,
) => {
	const submission = await prisma.submission.findUnique({
		where: { id },
		include: {
			problem: { select: { id: true, type: true, defaultMarks: true } },
			evaluation: { select: { reviewStatus: true, aiScore: true } },
			attempt: {
				select: {
					id: true,
					status: true,
					assessment: {
						select: {
							companyId: true,
							assessmentProblems: { select: { problemId: true, marks: true } },
						},
					},
				},
			},
		},
	});

	if (!submission) {
		throw new AppError(StatusCodes.NOT_FOUND, "Submission not found.");
	}

	assertCanGrade(
		companyId,
		submission.attempt.assessment.companyId,
		"RECRUITER",
	);

	if (
		submission.attempt.status === "NOT_STARTED" ||
		submission.attempt.status === "IN_PROGRESS"
	) {
		throw new AppError(
			StatusCodes.CONFLICT,
			"This attempt hasn't been submitted yet.",
		);
	}

	if (submission.problem.type === "MCQ") {
		throw new AppError(
			StatusCodes.BAD_REQUEST,
			"MCQ submissions are graded automatically and cannot be manually re-graded.",
		);
	}

	const assessmentProblem =
		submission.attempt.assessment.assessmentProblems.find(
			(ap) => ap.problemId === submission.problemId,
		);
	const maxScore = assessmentProblem?.marks ?? submission.problem.defaultMarks;

	if (payload.score > maxScore) {
		throw new AppError(
			StatusCodes.BAD_REQUEST,
			`Score cannot exceed the maximum marks for this problem (${maxScore}).`,
		);
	}

	const reviewStatus =
		submission.evaluation?.reviewStatus === "PENDING_REVIEW"
			? submission.evaluation.aiScore === payload.score
				? ("APPROVED" as const)
				: ("OVERRIDDEN" as const)
			: undefined;

	await prisma.$transaction(async (tx) => {
		if (submission.problem.type === "CODING" && payload.testCaseResults) {
			for (const testCaseResult of payload.testCaseResults) {
				await tx.testCaseResult.upsert({
					where: {
						submissionId_testCaseId: {
							submissionId: id,
							testCaseId: testCaseResult.testCaseId,
						},
					},
					update: {
						passed: testCaseResult.passed,
						...(testCaseResult.actualOutput !== undefined && {
							actualOutput: testCaseResult.actualOutput,
						}),
						points: testCaseResult.points ?? 0,
					},
					create: {
						submissionId: id,
						testCaseId: testCaseResult.testCaseId,
						passed: testCaseResult.passed,
						...(testCaseResult.actualOutput !== undefined && {
							actualOutput: testCaseResult.actualOutput,
						}),
						points: testCaseResult.points ?? 0,
					},
				});
			}
		}

		await tx.submission.update({
			where: { id },
			data: { status: "EVALUATED" },
		});

		await tx.submissionEvaluation.upsert({
			where: { submissionId: id },
			update: {
				score: payload.score,
				maxScore,
				status: "COMPLETED",
				isAutoEvaluated: false,
				evaluatorId,
				...(payload.feedback !== undefined && { feedback: payload.feedback }),
				...(reviewStatus && { reviewStatus, reviewedAt: new Date() }),
				evaluatedAt: new Date(),
			},
			create: {
				submissionId: id,
				score: payload.score,
				maxScore,
				status: "COMPLETED",
				isAutoEvaluated: false,
				evaluatorId,
				...(payload.feedback !== undefined && { feedback: payload.feedback }),
				evaluatedAt: new Date(),
			},
		});
	});

	await recomputeResult(submission.attempt.id);

	return prisma.submission.findUniqueOrThrow({
		where: { id },
		select: SUBMISSION_GRADING_SELECT,
	});
};

/**
 * Accepts the AI-suggested score and feedback as the final grade. Only a
 * recruiter can do this, so the human stays the final decision maker.
 */
const approveAiSuggestion = async (
	id: string,
	evaluatorId: string,
	companyId: string,
) => {
	const submission = await prisma.submission.findUnique({
		where: { id },
		include: {
			evaluation: true,
			attempt: {
				select: {
					id: true,
					status: true,
					assessment: { select: { companyId: true } },
				},
			},
		},
	});

	if (!submission) {
		throw new AppError(StatusCodes.NOT_FOUND, "Submission not found.");
	}

	assertCanGrade(
		companyId,
		submission.attempt.assessment.companyId,
		"RECRUITER",
	);

	const evaluation = submission.evaluation;

	if (
		!evaluation ||
		evaluation.reviewStatus !== "PENDING_REVIEW" ||
		evaluation.aiScore === null
	) {
		throw new AppError(
			StatusCodes.CONFLICT,
			"There is no AI suggestion waiting for review on this submission.",
		);
	}

	if (evaluation.status === "COMPLETED") {
		throw new AppError(
			StatusCodes.CONFLICT,
			"This submission has already been graded.",
		);
	}

	const aiScore = evaluation.aiScore;
	const aiFeedback = evaluation.aiFeedback;

	await prisma.$transaction([
		prisma.submission.update({
			where: { id },
			data: { status: "EVALUATED" },
		}),
		prisma.submissionEvaluation.update({
			where: { submissionId: id },
			data: {
				score: aiScore,
				...(aiFeedback !== null && { feedback: aiFeedback }),
				status: "COMPLETED",
				isAutoEvaluated: false,
				evaluatorId,
				reviewStatus: "APPROVED",
				reviewedAt: new Date(),
				evaluatedAt: new Date(),
			},
		}),
	]);

	await recomputeResult(submission.attempt.id);

	return prisma.submission.findUniqueOrThrow({
		where: { id },
		select: SUBMISSION_GRADING_SELECT,
	});
};

	getSubmissionsForAttempt,
	getSubmissionById,
	getPendingEvaluations,
	evaluateSubmission,
	approveAiSuggestion,
};

import { z } from "zod";

const mcqOptionSchema = z.object({
	optionText: z.string().trim().min(1, "Option text is required.").max(500),
	isCorrect: z.boolean(),
	order: z.number().int().min(1, "Option order must start at 1."),
});

const testCaseSchema = z.object({
	input: z.string().max(5000).optional(),
	expectedOutput: z
		.string()
		.trim()
		.min(1, "Expected output is required.")
		.max(5000),
	isSample: z.boolean().optional(),
	points: z.coerce.number().int().min(0).max(1000).optional(),
	timeLimitMs: z.coerce.number().int().positive().optional(),
	memoryLimitMb: z.coerce.number().int().positive().optional(),
});

const baseFields = {
	title: z
		.string()
		.trim()
		.min(3, "Title must be at least 3 characters.")
		.max(200),
	description: z
		.string()
		.trim()
		.min(10, "Description must be at least 10 characters.")
		.max(10000),
	difficulty: z.enum(["EASY", "MEDIUM", "HARD"]).optional(),
	defaultMarks: z.coerce
		.number()
		.int()
		.min(1, "Marks must be at least 1.")
		.max(1000)
		.optional(),
	isPublic: z.boolean().optional(),
};

const mcqCreateSchema = z.object({
	...baseFields,
	type: z.literal("MCQ"),
	mcqType: z
		.enum(["SINGLE_CHOICE", "MULTIPLE_CHOICE"])
		.default("SINGLE_CHOICE"),
	explanation: z.string().trim().max(2000).optional(),
	options: z
		.array(mcqOptionSchema)
		.min(2, "Provide at least 2 options.")
		.max(10, "At most 10 options are allowed."),
});

const codingCreateSchema = z.object({
	...baseFields,
	type: z.literal("CODING"),
	timeLimitSeconds: z.coerce.number().int().min(1).max(7200).optional(),
	testCases: z.array(testCaseSchema).min(1, "Provide at least 1 test case."),
});

const writtenCreateSchema = z.object({
	...baseFields,
	type: z.literal("WRITTEN"),
	rubric: z.string().trim().min(10).max(5000).optional(),
});

/**
 * The per-type shape (options required for MCQ, testCases for CODING) is
 * enforced by the discriminated union itself. The cross-field rules that
 * a union alone can't express - unique option order, at least one correct
 * option, single-choice having exactly one - are enforced in superRefine
 * below, which runs after the union has already picked a branch.
 */
export const createProblemSchema = z
	.discriminatedUnion("type", [
		mcqCreateSchema,
		codingCreateSchema,
		writtenCreateSchema,
	])
	.superRefine((data, ctx) => {
		if (data.type !== "MCQ") return;

		const orders = data.options.map((option) => option.order);
		if (new Set(orders).size !== orders.length) {
			ctx.addIssue({
				code: "custom",
				message: "Option order values must be unique.",
				path: ["options"],
			});
		}

		const correctCount = data.options.filter(
			(option) => option.isCorrect,
		).length;

		if (correctCount === 0) {
			ctx.addIssue({
				code: "custom",
				message: "At least one option must be marked correct.",
				path: ["options"],
			});
		}

		if (data.mcqType === "SINGLE_CHOICE" && correctCount > 1) {
			ctx.addIssue({
				code: "custom",
				message:
					"SINGLE_CHOICE questions must have exactly one correct option.",
				path: ["options"],
			});
		}
	});

/**
 * `type` can't change after creation (see problem.interface.ts). Nested
 * `options`/`testCases`, when provided, fully replace the existing set -
 * see problem.service.ts#updateProblem for how that's done safely against
 * the onDelete: Restrict foreign keys from SubmissionAnswer/TestCaseResult.
 */
export const updateProblemSchema = z
	.object({
		title: z.string().trim().min(3).max(200).optional(),
		description: z.string().trim().min(10).max(10000).optional(),
		rubric: z.string().trim().min(10).max(5000).optional(),
		difficulty: z.enum(["EASY", "MEDIUM", "HARD"]).optional(),
		defaultMarks: z.coerce.number().int().min(1).max(1000).optional(),
		isPublic: z.boolean().optional(),
		timeLimitSeconds: z.coerce.number().int().min(1).max(7200).optional(),
		mcqType: z.enum(["SINGLE_CHOICE", "MULTIPLE_CHOICE"]).optional(),
		explanation: z.string().trim().max(2000).optional(),
		options: z.array(mcqOptionSchema).min(2).max(10).optional(),
		testCases: z.array(testCaseSchema).min(1).optional(),
	})
	.superRefine((data, ctx) => {
		if (!data.options) return;

		const orders = data.options.map((option) => option.order);
		if (new Set(orders).size !== orders.length) {
			ctx.addIssue({
				code: "custom",
				message: "Option order values must be unique.",
				path: ["options"],
			});
		}

		if (!data.options.some((option) => option.isCorrect)) {
			ctx.addIssue({
				code: "custom",
				message: "At least one option must be marked correct.",
				path: ["options"],
			});
		}
	});

export const problemValidation = {
	createProblemSchema,
	updateProblemSchema,

};

const getProblemById = async (id: string, companyId?: string) => {
    // tenant-scoped via withTenantScope
    const whereClause = withTenantScope({ id, deletedAt: null }, companyId) as ProblemWhereInput;

    const problem = await prisma.problem.findFirst({
        where: whereClause,
        select: PROBLEM_DETAIL_SELECT,
    });

    if (!problem) {
        throw new AppError(StatusCodes.NOT_FOUND, "Problem not found.");
    }

    return problem;
};

/**
 * Updates top-level fields, and optionally fully replaces the nested
 * options/testCases set.
 */
const updateProblem = async (
    id: string,
    companyId: string,
    payload: UpdateProblemInput,
) => {
    // tenant-scoped via withTenantScope
    const whereClause = withTenantScope({ id, deletedAt: null }, companyId) as ProblemWhereInput;

    const existing = await prisma.problem.findFirst({
        where: whereClause,
    });

    if (!existing) {
        throw new AppError(StatusCodes.NOT_FOUND, "Problem not found.");
    }

    if (payload.rubric !== undefined && existing.type !== "WRITTEN") {
        throw new AppError(
            StatusCodes.BAD_REQUEST,
            "A rubric can only be set on a WRITTEN problem.",
        );
    }

    const { options, testCases, mcqType, explanation, ...topLevel } = payload;

    if (Object.keys(topLevel).length > 0) {
        await prisma.problem.update({ where: { id }, data: topLevel });
    }

    if (
        existing.type === "MCQ" &&
        (options || mcqType !== undefined || explanation !== undefined)
    ) {
        const mcqProblem = await prisma.mcqProblem.findUniqueOrThrow({
            where: { problemId: id },
        });

        if (mcqType !== undefined || explanation !== undefined) {
            await prisma.mcqProblem.update({
                where: { id: mcqProblem.id },
                data: {
                    ...(mcqType !== undefined && { type: mcqType }),
                    ...(explanation !== undefined && { explanation }),
                },
            });
        }

        if (options) {
            await prisma.$transaction(
                options.map((option) =>
                    prisma.mcqOption.upsert({
                        where: {
                            mcqProblemId_order: {
                                mcqProblemId: mcqProblem.id,
                                order: option.order,
                            },
                        },
                        update: {
                            optionText: option.optionText,
                            isCorrect: option.isCorrect,
                        },
                        create: { mcqProblemId: mcqProblem.id, ...option },
                    }),
                ),
            );
        }
    }

    if (existing.type === "CODING" && testCases) {
        const hasGradedResult = await prisma.testCaseResult.findFirst({
            where: { testCase: { problemId: id } },
        });

        if (hasGradedResult) {
            throw new AppError(
                StatusCodes.CONFLICT,
                "This problem's test cases can't be changed after candidates have been graded against them. Create a new problem instead.",
            );
        }

        await prisma.testCase.deleteMany({ where: { problemId: id } });
        await prisma.testCase.createMany({
            data: testCases.map((testCase) => ({ problemId: id, ...testCase })),
        });
    }

    return getProblemById(id, companyId);
};

/**
 * Soft delete.
 */
const softDeleteProblem = async (id: string, companyId: string) => {
    // tenant-scoped via withTenantScope
    const whereClause = withTenantScope({ id, deletedAt: null }, companyId) as ProblemWhereInput;

    const existing = await prisma.problem.findFirst({
        where: whereClause,
    });

    if (!existing) {
        throw new AppError(StatusCodes.NOT_FOUND, "Problem not found.");
    }

    const usedInLiveAssessment = await prisma.assessmentProblem.findFirst({
        where: {
            problemId: id,
            assessment: { status: { in: ["PUBLISHED", "ACTIVE"] }, deletedAt: null },
        },
    });

    if (usedInLiveAssessment) {
        throw new AppError(
            StatusCodes.CONFLICT,
            "This problem is part of a published assessment and can't be deleted. Remove it from the assessment first.",
        );
    }

    await prisma.problem.update({
        where: { id },
        data: { deletedAt: new Date() },
    });

    return { message: "Problem deleted successfully." };
};

export const problemService = {
    createProblem,
    getAllProblems,
    getProblemById,
    updateProblem,
    softDeleteProblem,
};

import type { Difficulty, McqType } from "../../../generated/prisma/enums";

export type McqOptionInput = {
	optionText: string;
	isCorrect: boolean;
	order: number;

export type TestCaseInput = {
	input?: string;
	expectedOutput: string;
	isSample?: boolean;
	points?: number;
	timeLimitMs?: number;
	memoryLimitMb?: number;
};

type BaseProblemFields = {
	title: string;
	description: string;
	difficulty?: Difficulty;
	defaultMarks?: number;
	isPublic?: boolean;
};

export type CreateProblemInput =
	| (BaseProblemFields & {
			type: "MCQ";
			mcqType?: McqType;
			explanation?: string;
			options: McqOptionInput[];
	  })
	| (BaseProblemFields & {
			type: "CODING";
			timeLimitSeconds?: number;
			testCases: TestCaseInput[];
	  })
	| (BaseProblemFields & { type: "WRITTEN"; rubric?: string });

/**
 * `type` is intentionally excluded - switching a problem from MCQ to
 * CODING (or similar) after creation would orphan its nested data, so it's
 * treated as immutable. Create a new problem instead.
 * `rubric` only applies to WRITTEN problems.
 */
export type UpdateProblemInput = Partial<{
	title: string;
	description: string;
	rubric: string;
	difficulty: Difficulty;
	defaultMarks: number;
	isPublic: boolean;
	timeLimitSeconds: number;
	mcqType: McqType;
	explanation: string;
	options: McqOptionInput[];
	testCases: TestCaseInput[];
}>;
